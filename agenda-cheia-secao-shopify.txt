(function () {
  function init(root) {
    if (!root || root.__acInit) return;
    root.__acInit = true;

    /* Barra fixa: aparece quando hero, oferta, CTA final e rodapé estão fora da tela */
    var sticky = root.querySelector('[data-ac-sticky]');
    if (sticky && 'IntersectionObserver' in window) {
      var seen = new Set();
      var io = new IntersectionObserver(function (entries) {
        entries.forEach(function (e) {
          if (e.isIntersecting) seen.add(e.target); else seen.delete(e.target);
        });
        sticky.classList.toggle('is-visible', seen.size === 0);
      });
      root.querySelectorAll('[data-ac-hide-sticky]').forEach(function (el) { io.observe(el); });
    }

    /* Eventos opcionais (desligados por padrão para evitar duplicidade com os apps da Shopify) */
    if (root.getAttribute('data-ac-track') !== 'true') return;
    var id = root.getAttribute('data-item-id');
    if (!id) return;
    var item = {
      item_id: id,
      item_name: root.getAttribute('data-item-name'),
      price: parseFloat(root.getAttribute('data-price')) || 0,
      quantity: 1
    };
    var currency = root.getAttribute('data-currency') || 'BRL';

    function gaEvent(name) {
      if (typeof window.gtag === 'function') {
        window.gtag('event', name, { currency: currency, value: item.price, items: [item] });
      }
    }
    function fbEvent(name) {
      if (typeof window.fbq === 'function') {
        window.fbq('track', name, { content_ids: [id], content_type: 'product', content_name: item.item_name, value: item.price, currency: currency });
      }
    }

    gaEvent('view_item');
    fbEvent('ViewContent');

    var form = root.querySelector('[data-ac-form]');
    if (form) {
      form.addEventListener('submit', function () {
        gaEvent('add_to_cart');
        fbEvent('AddToCart');
      });
    }
  }

  function boot() { document.querySelectorAll('[data-ac-root]').forEach(init); }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot); else boot();
  document.addEventListener('shopify:section:load', boot);
})();
